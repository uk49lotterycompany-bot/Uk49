CREATE OR REPLACE FUNCTION public.admin_manage_withdrawal( target_request_id UUID, admin_action TEXT ) RETURNS JSONB LANGUAGE plpgsql SECURITY DEFINER SET search_path TO 'public' AS $function$ DECLARE request_record public.withdrawal_requests%ROWTYPE; wallet_record public.member_wallets%ROWTYPE; fee_record public.withdrawal_fee_payments%ROWTYPE; admin_id UUID; withdrawal_reference TEXT; BEGIN -- Confirm administrator identity. admin_id := auth.uid(); IF admin_id IS NULL OR NOT EXISTS ( SELECT 1 FROM public.admin_users WHERE user_id = admin_id ) THEN RAISE EXCEPTION 'Administrator access required.'; END IF; IF admin_action NOT IN ('approve', 'reject', 'complete') THEN RAISE EXCEPTION 'Invalid withdrawal action.'; END IF; -- Lock the withdrawal request. SELECT * INTO request_record FROM public.withdrawal_requests WHERE id = target_request_id FOR UPDATE; IF NOT FOUND THEN RAISE EXCEPTION 'Withdrawal request not found.'; END IF; -- Reject request. IF admin_action = 'reject' THEN IF request_record.status NOT IN ('pending', 'processing') THEN RAISE EXCEPTION 'This request cannot be rejected.'; END IF; UPDATE public.withdrawal_requests SET status = 'rejected', updated_at = NOW() WHERE id = target_request_id; RETURN jsonb_build_object( 'success', true, 'message', 'Withdrawal request rejected.' ); END IF; -- Require verified separate fee payment. SELECT * INTO fee_record FROM public.withdrawal_fee_payments WHERE withdrawal_request_id = target_request_id AND member_id = request_record.member_id FOR UPDATE; IF NOT FOUND THEN RAISE EXCEPTION 'Separate withdrawal fee payment has not been submitted.'; END IF; IF fee_record.payment_status <> 'verified' THEN RAISE EXCEPTION 'The separate withdrawal fee payment must be verified first.'; END IF; IF fee_record.fee_amount <> request_record.withdrawal_fee THEN RAISE EXCEPTION 'Verified fee amount does not match the withdrawal fee.'; END IF; -- Approve request. IF admin_action = 'approve' THEN IF request_record.status <> 'pending' THEN RAISE EXCEPTION 'Only pending requests can be approved.'; END IF; UPDATE public.withdrawal_requests SET status = 'processing', updated_at = NOW() WHERE id = target_request_id; RETURN jsonb_build_object( 'success', true, 'message', 'Fee verified. Withdrawal moved to processing.' ); END IF; -- Complete withdrawal. IF admin_action = 'complete' THEN IF request_record.status <> 'processing' THEN RAISE EXCEPTION 'Only processing withdrawals can be completed.'; END IF; withdrawal_reference := 'withdrawal:' || target_request_id::TEXT; -- Check whether this withdrawal was already recorded. IF EXISTS ( SELECT 1 FROM public.wallet_transactions WHERE reference = withdrawal_reference ) THEN RAISE EXCEPTION 'This withdrawal already has a wallet transaction. No additional debit was made.'; END IF; -- Lock member wallet. SELECT * INTO wallet_record FROM public.member_wallets WHERE id = request_record.wallet_id FOR UPDATE; IF NOT FOUND THEN RAISE EXCEPTION 'Member wallet not found.'; END IF; -- Check balance. IF wallet_record.balance < request_record.amount THEN RAISE EXCEPTION 'Insufficient wallet balance to complete withdrawal.'; END IF; -- Deduct the full withdrawal amount. UPDATE public.member_wallets SET balance = balance - request_record.amount, updated_at = NOW() WHERE id = request_record.wallet_id; -- Record the debit exactly once. INSERT INTO public.wallet_transactions ( wallet_id, transaction_type, amount, currency_code, description, reference ) VALUES ( request_record.wallet_id, 'withdrawal', request_record.amount, wallet_record.currency_code, 'Completed withdrawal. Request ID: ' || target_request_id::TEXT, withdrawal_reference ); -- Mark request completed. UPDATE public.withdrawal_requests SET status = 'completed', updated_at = NOW() WHERE id = target_request_id; RETURN jsonb_build_object( 'success', true, 'message', 'Withdrawal completed successfully.', 'amount_debited', request_record.amount, 'separately_paid_fee', fee_record.fee_amount, 'member_payout', request_record.amount, 'currency', wallet_record.currency_code ); END IF; END; $function$;
